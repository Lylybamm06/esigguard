"""Auth Service v3 - Affichage amélioré"""
from flask import Flask, jsonify
import os, sys, requests
from datetime import datetime, timezone

sys.path.insert(0, '/app')
from database.database import get_db

app = Flask(__name__)
WHOIS_API_KEY = os.getenv("WHOIS_API_KEY")
WHOIS_URL = "https://www.whoisxmlapi.com/whoisserver/WhoisService"

def get_domain_age(domain):
    if not WHOIS_API_KEY:
        return {"status": "unknown", "age_years": None}
    try:
        r = requests.get(f"{WHOIS_URL}?apiKey={WHOIS_API_KEY}&domainName={domain}&outputFormat=JSON", timeout=10)
        created = r.json().get("WhoisRecord", {}).get("createdDate")
        if not created:
            return {"status": "unknown", "age_years": None}
        age_years = (datetime.now(timezone.utc) - datetime.fromisoformat(created.replace("Z", "+00:00"))).days / 365
        return {"age_years": round(age_years, 2), "status": "bon" if age_years >= 1 else "suspect"}
    except:
        return {"status": "unknown", "age_years": None}

def analyze_auth(analysis_id):
    db = get_db()
    a = db.get_complete_analysis(analysis_id)
    if not a:
        raise ValueError("Introuvable")
    
    from_addr = a.get('email_sender', '')
    from_domain = from_addr.split('@')[-1] if '@' in from_addr else ''
    display = a.get('display_name', '')
    reply = a.get('reply_to', '')
    return_path = a.get('return_path', '')
    
    spf, dkim, dmarc = a.get('auth_spf','unknown'), a.get('auth_dkim','unknown'), a.get('auth_dmarc','unknown')
    age = get_domain_age(from_domain)
    
    score, issues = 0, []
    
    # SPF
    if spf != "pass":
        score += 25
        issues.append(f"SPF:{spf}")
    
    # DKIM
    if dkim != "pass":
        score += 25
        issues.append(f"DKIM:{dkim}")
    
    # DMARC
    if dmarc != "pass":
        score += 25
        issues.append(f"DMARC:{dmarc}")
    
    # From vs Return-Path (comparaison domaines)
    return_domain = return_path.split('@')[-1] if '@' in return_path and return_path else from_domain
    if from_domain != return_domain:
        score += 30
        issues.append("FromVsReturn: Pas coherent")
    else:
        issues.append("FromVsReturn: Coherent")
    
    # From vs Reply-To (comparaison domaines)
    if reply:
        reply_domain = reply.split('@')[-1] if '@' in reply else from_domain
        if from_domain != reply_domain:
            score += 30
            issues.append("FromVsReply: Pas coherent")
        else:
            issues.append("FromVsReply: Coherent")
    
    # Display Name vs Domain
    if display:
        # Vérifier si le domaine ou une partie du domaine est dans le display name
        domain_base = from_domain.split('.')[0].lower()  # ex: "google" de "google.com"
        if domain_base not in display.lower():
            score += 40
            issues.append("DisplayNameVsDomain: Pas coherent")
        else:
            issues.append("DisplayNameVsDomain: Coherent")
    
    # Âge du domaine
    if age.get("status") == "suspect":
        score += 25
        issues.append(f"AgeDomaine: Recent ({age.get('age_years')}ans)")
    elif age.get("status") == "unknown":
        score += 10
        issues.append("AgeDomaine: Inconnu")
    else:
        issues.append(f"AgeDomaine: OK ({age.get('age_years')}ans)")
    
    explanation = "Authentification : " + " | ".join(issues) if issues else "Authentification : Tout OK"
    
    return {
        "score": int((score/200)*100), 
        "explanation": explanation
    }

@app.route('/')
def home():
    return jsonify({"service": "Auth", "port": 5003})

@app.route('/health')
def health():
    try:
        db = get_db()
        db.get_connection().close()
        return jsonify({"status": "healthy"})
    except:
        return jsonify({"status": "unhealthy"}), 500

@app.post("/analyze/<int:analysis_id>")
def analyze(analysis_id):
    try:
        return jsonify({"analysis_id": analysis_id, "service": "auth", **analyze_auth(analysis_id)})
    except Exception as e:
        return jsonify({"error": str(e)}), 500

if __name__ == "__main__":
    print("\n🔐 Auth Service - Port 5003")
    print("Affichage amélioré : FromVsReturn, FromVsReply, DisplayNameVsDomain")
    app.run(host="0.0.0.0", port=5003, debug=True)
